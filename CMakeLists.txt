cmake_minimum_required(VERSION 3.27)
project(hypr-wmdeco VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
find_package(PkgConfig REQUIRED)
pkg_check_modules(deps REQUIRED IMPORTED_TARGET hyprland libdrm libinput libudev pixman-1 wayland-server xkbcommon pangocairo)
add_library(hypr-wmdeco SHARED src/main.cpp src/deco.cpp src/paint.cpp)
target_link_libraries(hypr-wmdeco PRIVATE PkgConfig::deps)
target_compile_options(hypr-wmdeco PRIVATE -Wall -Wextra -Wno-unused-parameter)
install(TARGETS hypr-wmdeco LIBRARY DESTINATION lib)
install(FILES README.md SOURCES.md TESTING.md LICENSE LICENSE.hyprbars LICENSE.wraster DESTINATION share/doc/hypr-wmdeco)
